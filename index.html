function onResults(results) { // 1. 繪製清晰的原圖（作為底層，保留頭髮與背景細節） canvasCtx.globalCompositeOperation = 'source-over'; canvasCtx.filter = 'none'; canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height); if (results.multiFaceLandmarks && results.multiFaceLandmarks.length > 0) { const landmarks = results.multiFaceLandmarks[0]; // 2. 建立遮罩範圍（暫以鼻樑為中心，建立橢圓形皮膚遮罩） canvasCtx.save(); canvasCtx.beginPath(); // 取得鼻樑點位 (index 1) 的真實像素座標 const nose = landmarks[1]; const x = nose.x * canvasElement.width; const y = nose.y * canvasElement.height; // 畫一個覆蓋臉頰的橢圓 canvasCtx.ellipse(x, y, 100, 140, 0, 0, 2 * Math.PI); canvasCtx.clip(); // 關鍵：接下來的繪圖只會出現在這個橢圓內 // 3. 在遮罩範圍內，疊加「模糊與提亮」的柔焦圖層 canvasCtx.filter = 'blur(6px) brightness(1.15)'; canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height); canvasCtx.restore(); // 解除遮罩，以免影響下一幀 } }
